#include <Arduino.h>
#include <Wire.h>
#include "DFRobot_EOxygenSensor.h"

#define I2C_SDA 21
#define I2C_SCL 22
#define OXYGEN_I2C_ADDRESS E_OXYGEN_ADDRESS_0

DFRobot_EOxygenSensor_I2C oxygen(&Wire, OXYGEN_I2C_ADDRESS);

void setup()
{
  Serial.begin(115200);
  while(!Serial); 

  Wire.begin(I2C_SDA, I2C_SCL);

  while(!oxygen.begin()){
    Serial.println("No devices found!");
    delay(1000);
  } 
  Serial.println("Device connected successfully!");

  // ==========================================
  // CALIBRATION SEQUENCE
  // ==========================================
  Serial.println("Starting calibration in 5 seconds... Please ensure sensor is in normal room air (20.9% O2).");
  delay(5000);

  // 1. Clear any old, faulty calibration data
  if(oxygen.clearCalibration()) {
    Serial.println("Old calibration data cleared successfully!");
  } else {
    Serial.println("Failed to clear calibration.");
  }
  delay(1000);

  // 2. Set the current environment as 20.9% Oxygen
  if(oxygen.calibration_20_9()) {
    Serial.println("20.9% VOL Single-Point Calibration successful!");
  } else {
    Serial.println("Calibration failed.");
  }
  delay(1000);
  // ==========================================
}

void loop()
{
  float o2_concentration = oxygen.readOxygenConcentration();
  
  Serial.print("Oxygen concentration is ");
  Serial.print(o2_concentration);
  Serial.println("% VOL");
  
  delay(1000);
}
